<objective>
Build the three-step permission walkthrough that replaces the OnboardingPlaceholderScreen. Each step shows a priming screen explaining the permission, a button to open the correct system Settings, ON_RESUME re-check that shows a green checkmark after grant, and skip/continue navigation. A step counter and progress bar are visible throughout.

Purpose: This is the core onboarding UX for new users. Without guided permission granting, users cannot use the app's primary features. The walkthrough turns three separate system Settings actions into a coherent, understandable flow.
Output: Working PermissionWalkthroughScreen with ViewModel, enum, and Settings launch helpers -- wired into the nav graph in place of OnboardingPlaceholderScreen.
</objective>

   ```kotlin
   package com.usagecontrol.feature.onboarding

   import android.content.ComponentName
   import android.content.Context
   import android.service.notification.NotificationListenerService
   import androidx.lifecycle.ViewModel
   import androidx.lifecycle.viewModelScope
   import com.usagecontrol.core.common.PermissionChecker
   import com.usagecontrol.core.common.onboarding.OnboardingRepository
   import dagger.hilt.android.lifecycle.HiltViewModel
   import dagger.hilt.android.qualifiers.ApplicationContext
   import kotlinx.coroutines.flow.MutableStateFlow
   import kotlinx.coroutines.flow.StateFlow
   import kotlinx.coroutines.flow.asStateFlow
   import kotlinx.coroutines.flow.update
   import kotlinx.coroutines.launch
   import javax.inject.Inject

   data class PermissionWalkthroughState(
       val currentStep: Int = 0,
       val permissionStates: List<Boolean> = List(PermissionStep.TOTAL_STEPS) { false },
       val isComplete: Boolean = false,
   )

   @HiltViewModel
   class PermissionWalkthroughViewModel @Inject constructor(
       private val permissionChecker: PermissionChecker,
       private val onboardingRepository: OnboardingRepository,
       @ApplicationContext private val context: Context,
   ) : ViewModel() {

       private val _state = MutableStateFlow(PermissionWalkthroughState())
       val state: StateFlow<PermissionWalkthroughState> = _state.asStateFlow()

       init {
           recheckAll()
           // Auto-advance past already-granted steps (handles app kill + relaunch mid-walkthrough)
           autoAdvancePastGranted()
       }

       /** Called from ON_RESUME observer -- re-reads all three permissions. */
       fun recheckAll() {
           val wasNotificationGranted = _state.value.permissionStates[PermissionStep.NotificationAccess.index]
           val isNowNotificationGranted = permissionChecker.isNotificationAccessGranted()

           // Pitfall 5: NLS rebind after Notification Access grant transition
           if (!wasNotificationGranted && isNowNotificationGranted) {
               try {
                   NotificationListenerService.requestRebind(
                       ComponentName(
                           context.packageName,
                           "com.usagecontrol.service.notification.NotificationCaptureService",
                       )
                   )
               } catch (_: Exception) {
                   // Safe to ignore -- requestRebind may fail if component not declared
               }
           }

           _state.update { current ->
               current.copy(
                   permissionStates = listOf(
                       isNowNotificationGranted,
                       permissionChecker.isUsageAccessGranted(),
                       permissionChecker.isDndAccessGranted(),
                   )
               )
           }
       }

       /** Advance to next step (after grant or skip). */
       fun advanceStep() {
           val nextStep = _state.value.currentStep + 1
           if (nextStep >= PermissionStep.TOTAL_STEPS) {
               completeOnboarding()
           } else {
               _state.update { it.copy(currentStep = nextStep) }
           }
       }

       private fun autoAdvancePastGranted() {
           // Skip already-granted steps on init (e.g., app killed and relaunched mid-walkthrough)
           val states = _state.value.permissionStates
           var step = _state.value.currentStep
           while (step < PermissionStep.TOTAL_STEPS && states[step]) {
               step++
           }
           if (step >= PermissionStep.TOTAL_STEPS) {
               completeOnboarding()
           } else if (step != _state.value.currentStep) {
               _state.update { it.copy(currentStep = step) }
           }
       }

       private fun completeOnboarding() {
           viewModelScope.launch {
               onboardingRepository.setOnboardingComplete(true)
               _state.update { it.copy(isComplete = true) }
           }
       }
   }
   ```

   Key design decisions:
   - `recheckAll()` detects NLS grant transition and calls `requestRebind()` (Pitfall 5 from research). Uses the string form of the ComponentName to avoid importing the service-notification module.
   - `autoAdvancePastGranted()` skips steps where permissions are already granted (Open Question 1 from research).
   - `completeOnboarding()` writes DataStore flag AND sets `isComplete` in state. The composable observes `isComplete` and calls the nav `onComplete` lambda. This means `MainViewModel.markOnboardingComplete()` is no longer called from `MainActivity` -- the `PermissionWalkthroughViewModel` handles it (research note: Option (a) is cleaner).
   - No user-facing strings in the ViewModel -- only step index and booleans. The composable maps step index to display content (anti-pattern avoidance from research).
   - `@ApplicationContext` is needed for `requestRebind` ComponentName -- this is a system API call, not a UI action, so it is acceptable in ViewModel (research Pitfall 4 note).

   import android.content.ComponentName
   import android.content.Context
   import android.content.Intent

   import android.os.Build
   import android.provider.Settings

   import androidx.compose.material3.*
   import androidx.compose.runtime.*
   import androidx.compose.ui.Alignment
   import androidx.compose.ui.Modifier
   import androidx.compose.ui.platform.LocalContext
   import androidx.compose.ui.text.style.TextAlign
   import androidx.compose.ui.unit.dp
   import androidx.hilt.navigation.compose.hiltViewModel
   import androidx.lifecycle.Lifecycle
   import androidx.lifecycle.LifecycleEventObserver
   import androidx.lifecycle.compose.LocalLifecycleOwner
   import androidx.lifecycle.compose.collectAsStateWithLifecycle

   @Composable
   fun PermissionWalkthroughScreen(
       onComplete: () -> Unit,
       viewModel: PermissionWalkthroughViewModel = hiltViewModel(),
   ) {
       val state by viewModel.state.collectAsStateWithLifecycle()
       val lifecycleOwner = LocalLifecycleOwner.current
       val context = LocalContext.current

       // ON_RESUME permission re-check (Pattern 2 from research)
       DisposableEffect(lifecycleOwner) {
           val observer = LifecycleEventObserver { _, event ->
               if (event == Lifecycle.Event.ON_RESUME) {
                   viewModel.recheckAll()
               }
           }
           lifecycleOwner.lifecycle.addObserver(observer)
           onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
       }

       // Navigate to main app when walkthrough completes
       LaunchedEffect(state.isComplete) {
           if (state.isComplete) onComplete()
       }

       PermissionWalkthroughContent(
           state = state,
           onOpenSettings = { step ->
               when (step) {
                   PermissionStep.NotificationAccess -> context.openNotificationListenerSettings()
                   PermissionStep.UsageAccess -> context.openUsageAccessSettings()
                   PermissionStep.DndAccess -> context.openDndAccessSettings()
               }
           },
           onContinue = { viewModel.advanceStep() },
           onSkip = { viewModel.advanceStep() },
       )
   }

   @Composable
   private fun PermissionWalkthroughContent(
       state: PermissionWalkthroughState,
       onOpenSettings: (PermissionStep) -> Unit,
       onContinue: () -> Unit,
       onSkip: () -> Unit,

       Scaffold { innerPadding ->

               Spacer(modifier = Modifier.height(32.dp))

               // Step content (PERM-01, PERM-03)

   /** Open Notification Listener Settings. API 30+: opens directly to this app's toggle. */
   private fun Context.openNotificationListenerSettings() {
       try {
           if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
               startActivity(Intent(Settings.ACTION_NOTIFICATION_LISTENER_DETAIL_SETTINGS).apply {
                   putExtra(
                       Settings.EXTRA_NOTIFICATION_LISTENER_COMPONENT_NAME,
                       ComponentName(
                           packageName,
                           "com.usagecontrol.service.notification.NotificationCaptureService",
                       ).flattenToString(),
                   )
               })
           } else {
               startActivity(Intent(Settings.ACTION_NOTIFICATION_LISTENER_SETTINGS))
           }
       } catch (_: Exception) {
           startActivity(Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
               data = Uri.fromParts("package", packageName, null)
           })
       }
   }

   /** Open Usage Access Settings. No package URI -- opens global list (Pitfall 3 from research). */
   private fun Context.openUsageAccessSettings() {
       try {
           startActivity(Intent(Settings.ACTION_USAGE_ACCESS_SETTINGS))
       } catch (_: Exception) {
           startActivity(Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
               data = Uri.fromParts("package", packageName, null)
           })
       }
   }

   /** Open DND Access Settings. No package URI -- opens global list. */
   private fun Context.openDndAccessSettings() {
       try {
           startActivity(Intent(Settings.ACTION_NOTIFICATION_POLICY_ACCESS_SETTINGS))
       } catch (_: Exception) {
           startActivity(Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
               data = Uri.fromParts("package", packageName, null)
           })
       }
   }
